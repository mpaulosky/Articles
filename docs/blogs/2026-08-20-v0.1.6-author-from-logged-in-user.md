## Overview

Version 0.1.6 integrates user authentication with article authorship. When creating or editing articles, the system automatically captures the logged-in user's information from Auth0 and assigns them as the article author, ensuring proper attribution and accountability.

## Technical Details

### User Context Service

New service to access current user information:

```csharp
public interface IUserContextService
{
    Task<User?> GetCurrentUserAsync();
    string? GetCurrentUserId();
    string? GetCurrentUserEmail();
    string? GetCurrentUserName();
}

public class UserContextService : IUserContextService
{
    private readonly IHttpContextAccessor _httpContextAccessor;

    public UserContextService(IHttpContextAccessor httpContextAccessor)
    {
        _httpContextAccessor = httpContextAccessor;
    }

    public string? GetCurrentUserId()
    {
        return _httpContextAccessor.HttpContext?.User
            .FindFirst(ClaimTypes.NameIdentifier)?.Value;
    }

    public string? GetCurrentUserEmail()
    {
        return _httpContextAccessor.HttpContext?.User
            .FindFirst(ClaimTypes.Email)?.Value;
    }

    public string? GetCurrentUserName()
    {
        return _httpContextAccessor.HttpContext?.User
            .FindFirst(ClaimTypes.Name)?.Value;
    }

    public async Task<User?> GetCurrentUserAsync()
    {
        var userId = GetCurrentUserId();
        if (string.IsNullOrEmpty(userId))
            return null;

        return await _userRepository.GetByIdAsync(userId);
    }
}
```

### Updated Create Command Handler

Automatic author assignment:

```csharp
public class CreateArticleHandler 
    : IRequestHandler<CreateArticleCommand, ArticleId>
{
    private readonly IArticleRepository _repository;
    private readonly IUserContextService _userContext;

    public async Task<ArticleId> Handle(
        CreateArticleCommand request,
        CancellationToken cancellationToken)
    {
        // Get current user automatically
        var userId = _userContext.GetCurrentUserId();
        var userName = _userContext.GetCurrentUserName();
        var userEmail = _userContext.GetCurrentUserEmail();

        if (string.IsNullOrEmpty(userId))
            throw new UnauthorizedException("User must be authenticated");

        // Create article with author information
        var article = Article.Create(
            request.Title,
            request.Content,
            request.CategoryId,
            userId,
            userName ?? "Unknown",
            userEmail ?? "unknown@example.com"
        );

        await _repository.AddAsync(article);
        return article.Id;
    }
}
```

### Authorization Policy

Users can only edit their own articles:

```csharp
public class ArticleAuthorizationHandler 
    : AuthorizationHandler<SameAuthorRequirement, Article>
{
    private readonly IUserContextService _userContext;

    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        SameAuthorRequirement requirement,
        Article article)
    {
        var currentUserId = _userContext.GetCurrentUserId();
        
        if (currentUserId == article.AuthorId)
        {
            context.Succeed(requirement);
        }

        return Task.CompletedTask;
    }
}

// Usage in controller/handler
var authResult = await _authorizationService
    .AuthorizeAsync(User, article, "SameAuthor");
    
if (!authResult.Succeeded)
    throw new UnauthorizedException("You can only edit your own articles");
```

UPDATE articles

    author_id = 'system',
    author_name = 'System',
    author_email = 'system@example.com',
    created_at = COALESCE(created_at, NOW())
