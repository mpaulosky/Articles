        // Assert

        _fileSystem.Setup(x => x.FileExists(It.IsAny<string>()))

        // Assert

    }

    [Fact]
    public async Task SaveAsync_Should_Create_Directory_If_Not_Exists()

public class LoggingBehaviorTests
{
    private readonly Mock<ILogger<LoggingBehavior<TestRequest, TestResponse>>> _logger;
    private readonly LoggingBehavior<TestRequest, TestResponse> _sut;

    [Fact]
    public async Task Should_Log_Request_Before_Handling()
    {
        // Arrange
        var request = new TestRequest();
        var next = () => Task.FromResult(new TestResponse());

        // Act
        await _sut.Handle(request, next, CancellationToken.None);

        // Assert
        _logger.Verify(x => x.Log(
            LogLevel.Information,
            It.IsAny<EventId>(),
            It.Is<It.IsAnyType>((v, t) => 
                v.ToString()!.Contains("Handling TestRequest")),
            null,
            It.IsAny<Func<It.IsAnyType, Exception?, string>>()), 
            Times.Once);
    }

    [Fact]
    public async Task Should_Log_Completion_Time()
    {
        // Arrange
        var request = new TestRequest();
        var next = async () =>
        {
            await Task.Delay(100);
            return new TestResponse();
        };

        // Act
        await _sut.Handle(request, next, CancellationToken.None);

        // Assert
        _logger.Verify(x => x.Log(
            LogLevel.Information,
            It.IsAny<EventId>(),
            It.Is<It.IsAnyType>((v, t) => 
                v.ToString()!.Contains("ElapsedMilliseconds")),
            null,
            It.IsAny<Func<It.IsAnyType, Exception?, string>>()), 
            Times.Once);
    }

    [Fact]
    public async Task Should_Log_Error_When_Handler_Throws()
    {
        // Arrange
        var request = new TestRequest();

        // Act

        // Assert

            LogLevel.Error,
            It.IsAny<EventId>(),
            It.Is<It.IsAnyType>((v, t) =>

            It.IsAny<Func<It.IsAnyType, Exception?, string>>()), 
            Times.Once);
    }

public class ValidationBehaviorTests
{
    private readonly Mock<IValidator<TestRequest>> _validator;
    private readonly ValidationBehavior<TestRequest, TestResponse> _sut;

    [Fact]
    public async Task Should_Call_Validators()
    {
        // Arrange
        var request = new TestRequest();
        var next = () => Task.FromResult(new TestResponse());
        _validator.Setup(x => x.ValidateAsync(
            It.IsAny<ValidationContext<TestRequest>>(),
            It.IsAny<CancellationToken>()))
            .ReturnsAsync(new ValidationResult());

        // Act
        await _sut.Handle(request, next, CancellationToken.None);

        // Assert
        _validator.Verify(x => x.ValidateAsync(
            It.Is<ValidationContext<TestRequest>>(c => c.InstanceToValidate == request),
            It.IsAny<CancellationToken>()), 
            Times.Once);
    }

    [Fact]
    public async Task Should_Throw_When_Validation_Fails()
    {
        // Arrange
        var request = new TestRequest();
        var next = () => Task.FromResult(new TestResponse());
        var failures = new List<ValidationFailure>
        {
            new ValidationFailure("Property", "Error message")
        };
        _validator.Setup(x => x.ValidateAsync(
            It.IsAny<ValidationContext<TestRequest>>(),
            It.IsAny<CancellationToken>()))
            .ReturnsAsync(new ValidationResult(failures));

        // Act
        var act = () => _sut.Handle(request, next, CancellationToken.None);

        // Assert
        await act.Should().ThrowAsync<ValidationException>()
            .Where(e => e.Errors.Any(f => f.ErrorMessage == "Error message"));
    }

    [Fact]
    public async Task Should_Continue_When_No_Validators()
    {
        // Arrange
        var sut = new ValidationBehavior<TestRequest, TestResponse>(
            Enumerable.Empty<IValidator<TestRequest>>());
        var request = new TestRequest();
        var expectedResponse = new TestResponse();
        var next = () => Task.FromResult(expectedResponse);

        // Act
        var result = await sut.Handle(request, next, CancellationToken.None);

        // Assert
        result.Should().Be(expectedResponse);
    }

## Build Warning Cleanup

### Categories Resolved

1. **Async Warnings (CA2007)**: Added `.ConfigureAwait(false)` where appropriate
2. **Nullable Warnings (CS8602)**: Improved null checking
3. **Unused Variables (CS0219)**: Removed or utilized
4. **Documentation (CS1591)**: Added XML documentation
5. **Dispose Patterns (CA1816)**: Implemented proper disposal

### Example Fixes

```csharp
// Before: CA2007 warning
public async Task<Article> GetArticleAsync(string id)
{
    return await _repository.GetByIdAsync(id);
}

// After: Configured await
public async Task<Article> GetArticleAsync(string id)
{
    return await _repository.GetByIdAsync(id)
        .ConfigureAwait(false);
}

// Before: CS8602 warning
public string GetAuthorName(Article article)
{
    return article.Author.Name; // Possible null reference
}

// After: Null checking
public string GetAuthorName(Article article)
{
    return article.Author?.Name ?? "Unknown";
}
```

## Testing Best Practices Demonstrated

### 1. Arrange-Act-Assert Pattern

Clear test structure:
```csharp
[Fact]
public async Task Should_Do_Something()
{
    // Arrange - Set up test data and mocks
    var input = CreateTestData();
    
    // Act - Execute the system under test
    var result = await _sut.PerformActionAsync(input);
    
    // Assert - Verify the outcome
    result.Should().BeExpectedValue();
}
```

### 2. Test Naming Convention

Descriptive test names:
- `Should_[ExpectedBehavior]_When_[Condition]`
- `Should_Throw_[Exception]_When_[InvalidCondition]`
- `Should_Return_[Value]_For_[Scenario]`

### 3. One Assert Per Test

Each test verifies one behavior:
```csharp
// Good - Single responsibility
[Fact]
public async Task Should_Save_File()
{
    await _sut.SaveAsync("file.txt", "content");
    _fileSystem.Verify(x => x.WriteAllTextAsync(...), Times.Once);
}

// Not ideal - Multiple responsibilities
[Fact]
public async Task Should_Save_And_Read_File()
{
    await _sut.SaveAsync("file.txt", "content");
    var result = await _sut.ReadAsync("file.txt");
    result.Should().Be("content");
}
```
